# 手写数字识别

## 目标

用 PyTorch 多层感知机（Multilayer Perceptron，MLP）识别 MNIST 0–9 手写数字，完整执行：

```text
下载与核验数据 → DataLoader → 前向传播
→ 交叉熵 → 反向传播 → 测试集评测
→ 混淆矩阵 → 错误样本索引 → 保存模型
```

## 数据

MNIST 包含 28×28 灰度手写数字图像。`torchvision.datasets.MNIST` 可下载训练集和测试集。测试集只能用于最终评测，不应反复调参；正式实验还应从训练数据中划出验证集。

## 环境与验证边界

需要 Python 3.10+、PyTorch 2.x、torchvision。安装方式以官方页面为准：https://pytorch.org/get-started/locally/

当前知识库所在机器未安装 PyTorch 和 torchvision，因此以下代码只完成语法检查，未在本机下载 MNIST 或运行训练。

## 完整单文件程序

保存为 `mnist_mlp.py`：

```python
import argparse
import json
import random
from pathlib import Path

import torch
from torch import nn
from torch.utils.data import DataLoader, Dataset, Subset, random_split
from torchvision import datasets, transforms


def set_seed(seed):
    random.seed(seed)
    torch.manual_seed(seed)
    if torch.cuda.is_available():
        torch.cuda.manual_seed_all(seed)


class IndexedDataset(Dataset):
    def __init__(self, dataset):
        self.dataset = dataset

    def __len__(self):
        return len(self.dataset)

    def __getitem__(self, index):
        image, label = self.dataset[index]
        original_index = index
        if isinstance(self.dataset, Subset):
            original_index = int(self.dataset.indices[index])
        return image, label, original_index


class MLP(nn.Module):
    def __init__(self):
        super().__init__()
        self.net = nn.Sequential(
            nn.Flatten(),
            nn.Linear(28 * 28, 256),
            nn.ReLU(),
            nn.Dropout(0.1),
            nn.Linear(256, 10),
        )

    def forward(self, images):
        return self.net(images)


@torch.no_grad()
def evaluate(model, loader, loss_fn, device):
    model.eval()
    loss_sum = 0.0
    correct = 0
    total = 0
    confusion = torch.zeros(10, 10, dtype=torch.long)
    mistakes = []

    for images, labels, original_indices in loader:
        images = images.to(device)
        labels = labels.to(device)
        logits = model(images)
        loss_sum += loss_fn(logits, labels).item() * labels.size(0)
        predictions = logits.argmax(dim=1)
        correct += (predictions == labels).sum().item()
        total += labels.numel()

        for truth, prediction in zip(labels.cpu(), predictions.cpu()):
            confusion[truth, prediction] += 1
        if len(mistakes) < 20:
            wrong = torch.nonzero(predictions != labels).flatten().tolist()
            for index in wrong:
                if len(mistakes) == 20:
                    break
                mistakes.append(
                    {
                        "dataset_index": int(original_indices[index]),
                        "truth": int(labels[index].item()),
                        "prediction": int(predictions[index].item()),
                    }
                )

    return {
        "loss": loss_sum / total,
        "accuracy": correct / total,
        "confusion": confusion.tolist(),
        "mistakes": mistakes,
    }


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--data", default="data")
    parser.add_argument("--output", default="mnist_mlp.pt")
    parser.add_argument("--epochs", type=int, default=3)
    parser.add_argument("--batch-size", type=int, default=64)
    parser.add_argument("--seed", type=int, default=7)
    args = parser.parse_args()

    set_seed(args.seed)
    device = "cuda" if torch.cuda.is_available() else "cpu"
    transform = transforms.ToTensor()
    full_train = datasets.MNIST(
        args.data, train=True, download=True, transform=transform
    )
    test_set = datasets.MNIST(
        args.data, train=False, download=True, transform=transform
    )

    generator = torch.Generator().manual_seed(args.seed)
    train_set, validation_set = random_split(
        full_train, [55000, 5000], generator=generator
    )
    train_loader = DataLoader(
        IndexedDataset(train_set),
        batch_size=args.batch_size,
        shuffle=True,
        generator=generator,
    )
    validation_loader = DataLoader(
        IndexedDataset(validation_set), batch_size=256, shuffle=False
    )
    test_loader = DataLoader(
        IndexedDataset(test_set), batch_size=256, shuffle=False
    )

    model = MLP().to(device)
    optimizer = torch.optim.AdamW(model.parameters(), lr=1e-3)
    loss_fn = nn.CrossEntropyLoss()

    print("device:", device)
    for epoch in range(args.epochs):
        model.train()
        train_loss_sum = 0.0
        train_count = 0
        for images, labels, _ in train_loader:
            images = images.to(device)
            labels = labels.to(device)
            logits = model(images)
            loss = loss_fn(logits, labels)
            optimizer.zero_grad(set_to_none=True)
            loss.backward()
            optimizer.step()
            train_loss_sum += loss.item() * labels.size(0)
            train_count += labels.size(0)

        validation = evaluate(model, validation_loader, loss_fn, device)
        print(
            f"epoch {epoch + 1} | train loss {train_loss_sum / train_count:.4f} | "
            f"val loss {validation['loss']:.4f} | "
            f"val accuracy {validation['accuracy']:.4f}"
        )

    test = evaluate(model, test_loader, loss_fn, device)
    print("test loss:", round(test["loss"], 4))
    print("test accuracy:", round(test["accuracy"], 4))
    print("confusion matrix:")
    for row in test["confusion"]:
        print(row)

    torch.save(
        {
            "model": model.state_dict(),
            "seed": args.seed,
            "torch_version": torch.__version__,
            "torchvision_version": __import__("torchvision").__version__,
            "test_metrics": test,
        },
        args.output,
    )
    Path(args.output + ".json").write_text(
        json.dumps(test, ensure_ascii=False, indent=2), encoding="utf-8"
    )
    print("saved:", args.output)


if __name__ == "__main__":
    main()
```

## 运行

```bash
python mnist_mlp.py --epochs 3
```

首次运行会联网下载 MNIST。之后能否离线复用取决于 `data/` 是否完整保留。

## 验收

- 输出训练平均 loss，而不是只打印最后一个 batch；
- 每轮报告独立验证集 loss 和 accuracy；
- 训练结束后只对测试集做最终报告；
- 测试准确率应明显高于随机猜测的 10%；类似 MLP 在该配置附近常能超过 95%，但具体结果取决于版本、设备、随机性和训练设置 `[C]`；
- 打印 10×10 混淆矩阵，抽查哪些数字容易混淆；
- `mnist_mlp.pt.json` 中保存最多 20 个错误样本的 `dataset_index`、真实标签和预测标签；对于测试集，该索引可直接用于 `test_set[dataset_index]` 重新读取原图；
- 保存 PyTorch、torchvision 版本和随机种子。

达到某个 accuracy 不证明模型适用于真实手写表单。MNIST 图像干净、居中且任务范围窄。

## 读懂混淆矩阵

行表示真实数字，列表示预测数字。若第 5 行第 3 列数值较大，表示真实数字 5 经常被预测为 3。平均准确率可能掩盖某些类别的集中错误。

## 常见失败

- 下载失败：检查网络和 torchvision 数据目录，不要从不明链接下载；
- CUDA 内存不足：减小 batch 或使用 CPU；
- loss 不下降：检查标签、输入范围、学习率和 `model.train()`；
- 测试结果异常高：确认测试集未进入训练；
- 每次结果差异大：记录种子和确定性设置，但不同硬件仍可能有数值差异。
