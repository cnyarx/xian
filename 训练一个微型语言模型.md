# 训练一个微型语言模型

## 目标

在一小段有权使用的 UTF-8 文本上训练字符级 decoder-only 语言模型，完整走通：

```text
文本 → 词表 → 训练/验证切分 → 批量
→ Transformer → 交叉熵 → 优化器更新
→ 验证损失 → 检查点 → 自回归生成
```

这个项目用于理解训练闭环，不用于获得实用聊天能力。字符模型和小数据只能学习局部字符模式。

## 环境与验证边界

需要 Python 3.10+、PyTorch 2.x。安装命令以官方页面为准：https://pytorch.org/get-started/locally/

当前知识库所在机器未安装 PyTorch，因此代码已做语法级检查，但未在本机完成训练运行。实际运行者必须记录 Python、PyTorch、设备和随机种子。

## 准备输入

建立 `input.txt`，使用你有权训练的文本。为避免样本过少，建议至少几万字符。最小冒烟测试可重复一段自写文本：

```text
春天来了，风吹过树林。春天来了，鸟飞过河流。
```

不要把测试答案、隐私数据或无权使用的文本混入训练集。

## 完整单文件程序

保存为 `train_char_lm.py`。它复用了项目 03 的核心结构，但为保证本文件可以独立运行，保留完整模型代码。

```python
import argparse
import json
import math
import random
from dataclasses import asdict, dataclass
from pathlib import Path

import torch
import torch.nn.functional as F
from torch import nn


@dataclass
class Config:
    block_size: int = 64
    dim: int = 128
    num_heads: int = 4
    num_layers: int = 3
    ffn_hidden: int = 512
    dropout: float = 0.1
    batch_size: int = 32
    learning_rate: float = 3e-4
    max_steps: int = 1000
    eval_interval: int = 100
    eval_batches: int = 20
    seed: int = 7


class CausalSelfAttention(nn.Module):
    def __init__(self, dim: int, num_heads: int, block_size: int, dropout: float):
        super().__init__()
        if dim % num_heads != 0:
            raise ValueError("dim must be divisible by num_heads")
        self.num_heads = num_heads
        self.head_dim = dim // num_heads
        self.qkv = nn.Linear(dim, 3 * dim, bias=False)
        self.out = nn.Linear(dim, dim, bias=False)
        self.dropout = nn.Dropout(dropout)
        mask = torch.triu(torch.ones(block_size, block_size, dtype=torch.bool), diagonal=1)
        self.register_buffer("mask", mask, persistent=False)

    def forward(self, x):
        batch, time, dim = x.shape
        q, k, v = self.qkv(x).chunk(3, dim=-1)
        q = q.view(batch, time, self.num_heads, self.head_dim).transpose(1, 2)
        k = k.view(batch, time, self.num_heads, self.head_dim).transpose(1, 2)
        v = v.view(batch, time, self.num_heads, self.head_dim).transpose(1, 2)
        scores = q @ k.transpose(-2, -1) / math.sqrt(self.head_dim)
        scores = scores.masked_fill(self.mask[:time, :time], float("-inf"))
        weights = self.dropout(torch.softmax(scores, dim=-1))
        output = weights @ v
        output = output.transpose(1, 2).contiguous().view(batch, time, dim)
        return self.dropout(self.out(output))


class Block(nn.Module):
    def __init__(self, config: Config):
        super().__init__()
        self.norm1 = nn.LayerNorm(config.dim)
        self.attn = CausalSelfAttention(
            config.dim, config.num_heads, config.block_size, config.dropout
        )
        self.norm2 = nn.LayerNorm(config.dim)
        self.ffn = nn.Sequential(
            nn.Linear(config.dim, config.ffn_hidden),
            nn.GELU(),
            nn.Linear(config.ffn_hidden, config.dim),
            nn.Dropout(config.dropout),
        )

    def forward(self, x):
        x = x + self.attn(self.norm1(x))
        return x + self.ffn(self.norm2(x))


class CharLM(nn.Module):
    def __init__(self, vocab_size: int, config: Config):
        super().__init__()
        self.config = config
        self.token_embedding = nn.Embedding(vocab_size, config.dim)
        self.position_embedding = nn.Embedding(config.block_size, config.dim)
        self.blocks = nn.ModuleList([Block(config) for _ in range(config.num_layers)])
        self.final_norm = nn.LayerNorm(config.dim)
        self.lm_head = nn.Linear(config.dim, vocab_size, bias=False)
        self.apply(self._init_weights)
        self.lm_head.weight = self.token_embedding.weight

    @staticmethod
    def _init_weights(module):
        if isinstance(module, (nn.Linear, nn.Embedding)):
            nn.init.normal_(module.weight, mean=0.0, std=0.02)
        if isinstance(module, nn.Linear) and module.bias is not None:
            nn.init.zeros_(module.bias)

    def forward(self, token_ids, targets=None):
        batch, time = token_ids.shape
        if time > self.config.block_size:
            raise ValueError("sequence exceeds block_size")
        positions = torch.arange(time, device=token_ids.device)
        x = self.token_embedding(token_ids) + self.position_embedding(positions)[None]
        for block in self.blocks:
            x = block(x)
        logits = self.lm_head(self.final_norm(x))
        loss = None
        if targets is not None:
            loss = F.cross_entropy(logits.reshape(batch * time, -1), targets.reshape(-1))
        return logits, loss

    @torch.no_grad()
    def generate(self, token_ids, max_new_tokens, temperature=0.8, top_k=20):
        self.eval()
        for _ in range(max_new_tokens):
            context = token_ids[:, -self.config.block_size :]
            logits, _ = self(context)
            next_logits = logits[:, -1] / temperature
            k = min(top_k, next_logits.size(-1))
            threshold = torch.topk(next_logits, k).values[:, [-1]]
            next_logits = next_logits.masked_fill(next_logits < threshold, float("-inf"))
            probabilities = torch.softmax(next_logits, dim=-1)
            next_token = torch.multinomial(probabilities, 1)
            token_ids = torch.cat([token_ids, next_token], dim=1)
        return token_ids


def set_seed(seed: int):
    random.seed(seed)
    torch.manual_seed(seed)
    if torch.cuda.is_available():
        torch.cuda.manual_seed_all(seed)


def build_batch(data, block_size, batch_size, device):
    if len(data) <= block_size:
        raise ValueError("split must contain more tokens than block_size")
    starts = torch.randint(0, len(data) - block_size, (batch_size,))
    x = torch.stack([data[i : i + block_size] for i in starts])
    y = torch.stack([data[i + 1 : i + block_size + 1] for i in starts])
    return x.to(device), y.to(device)


@torch.no_grad()
def estimate_loss(model, train_data, val_data, config, device):
    model.eval()
    result = {}
    for name, data in (("train", train_data), ("val", val_data)):
        losses = []
        for _ in range(config.eval_batches):
            x, y = build_batch(data, config.block_size, config.batch_size, device)
            _, loss = model(x, y)
            losses.append(loss.item())
        result[name] = sum(losses) / len(losses)
    model.train()
    return result


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--input", default="input.txt")
    parser.add_argument("--output", default="char_lm_checkpoint.pt")
    parser.add_argument("--max-steps", type=int, default=1000)
    args = parser.parse_args()

    config = Config(max_steps=args.max_steps)
    set_seed(config.seed)
    device = "cuda" if torch.cuda.is_available() else "cpu"

    text = Path(args.input).read_text(encoding="utf-8")
    minimum_characters = 10 * (config.block_size + 1)
    if len(text) < minimum_characters:
        raise ValueError(
            f"input text is too short; provide at least {minimum_characters} characters "
            "so the 10% validation split exceeds block_size"
        )

    chars = sorted(set(text))
    stoi = {char: index for index, char in enumerate(chars)}
    itos = {index: char for char, index in stoi.items()}
    encoded = torch.tensor([stoi[char] for char in text], dtype=torch.long)

    split = int(0.9 * len(encoded))
    train_data = encoded[:split]
    val_data = encoded[split:]
    if len(val_data) <= config.block_size:
        raise ValueError("validation split is too short; use a longer input file")

    model = CharLM(len(chars), config).to(device)
    optimizer = torch.optim.AdamW(model.parameters(), lr=config.learning_rate)

    expected_random_loss = math.log(len(chars))
    print("device:", device)
    print("characters:", len(text), "vocab:", len(chars))
    print("random baseline ln(V):", round(expected_random_loss, 4))

    for step in range(config.max_steps + 1):
        if step % config.eval_interval == 0:
            losses = estimate_loss(model, train_data, val_data, config, device)
            print(
                f"step {step:4d} | train {losses['train']:.4f} | val {losses['val']:.4f}"
            )

        if step == config.max_steps:
            break

        x, y = build_batch(
            train_data, config.block_size, config.batch_size, device
        )
        _, loss = model(x, y)
        optimizer.zero_grad(set_to_none=True)
        loss.backward()
        torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=1.0)
        optimizer.step()

    checkpoint = {
        "model": model.state_dict(),
        "config": asdict(config),
        "stoi": stoi,
        "itos": itos,
        "source_file": str(Path(args.input).resolve()),
    }
    torch.save(checkpoint, args.output)
    Path(args.output + ".json").write_text(
        json.dumps({"config": asdict(config), "vocab": chars}, ensure_ascii=False, indent=2),
        encoding="utf-8",
    )

    start = torch.tensor([[encoded[0].item()]], device=device)
    generated = model.generate(start, max_new_tokens=200)
    generated_text = "".join(itos[index] for index in generated[0].tolist())
    print("\n--- generated sample ---")
    print(generated_text)
    print("checkpoint:", args.output)


if __name__ == "__main__":
    main()
```

## 运行

```bash
python train_char_lm.py --input input.txt --max-steps 1000
```

CPU 冒烟测试可先减少步数：

```bash
python train_char_lm.py --input input.txt --max-steps 10
```

## 预期输出怎样判断

输出应包含：

```text
device: ...
characters: ... vocab: ...
random baseline ln(V): ...
step    0 | train ... | val ...
...
checkpoint: char_lm_checkpoint.pt
```

不要要求每次得到固定 loss。应检查：

1. 第 0 步损失数量级是否接近 $\ln(V)$；权重共享和初始化会让它不精确相等；
2. 训练损失是否总体下降；
3. 验证损失是否在一段时间内下降；
4. 若训练继续下降而验证上升，可能过拟合；
5. 生成文本是否逐渐学会局部字符和标点模式；
6. `.pt` 和 `.json` 检查点文件是否生成。

## 训练/验证切分边界

本程序按文本前 90%/后 10% 切分。连续文本相邻片段可能高度相似，因此验证损失可能偏乐观。更严格的实验应按文档、时间或来源切分，并去除重复内容。

## 常见失败

- `validation split is too short`：增加输入文本或减小 `block_size`；
- loss 为 NaN：检查学习率、输入、设备和 PyTorch 版本；
- 生成永远重复：数据太少、过拟合或采样温度过低；
- 验证损失不降：数据分布差异、模型容量或训练步数不合适；
- CPU 很慢：减小 `dim`、`num_layers`、`batch_size` 和步数。
